# ---------------------
# BASIC I/O SETTINGS
# ---------------------
 
# File names only (not full paths). Files are read/written from the
# same directory where this script lives.

INPUT_FILENAME = "KickStartMyHeart.mp3"
OUTPUT_FILENAME = "KickStartMyHeart_CRT_tv.mp3"
 
# TARGET_SAMPLE_RATE controls how "narrow" and lo fi the sound is.
#  - Lower (8000–16000): more telephone/TV like, less high end detail.
#  - Higher (22050+): slightly clearer, less 1960s.
TARGET_SAMPLE_RATE = 16000
 
# ---------------------
# TV SPEAKER / EQ TUNING
# ---------------------
TV_SOUND = {
    # Overall loudness after processing (in dB). 0.0 keeps original peak.
    #  - Turn this up if your output is too quiet.
    #  - Turn this down if your output is too loud and clipping.
    "output_gain_db": -2.0,
 
    # Band pass window in Hz. Frequencies outside this range are strongly
    # attenuated to mimic the limited bandwidth of old TV speakers.
    #  - Increase band_low_hz for a thinner, more "radio" sound.
    #  - Decrease band_high_hz to make it more muffled / boxy.
    "band_low_hz": 150.0,
    "band_high_hz": 4500.0,
 
    # 3 band EQ inside the band pass window:
    #   LOW  : 0 – low_mid_split_hz
    #   MID  : low_mid_split_hz – mid_high_split_hz
    #   HIGH : mid_high_split_hz – band_high_hz
    #
    # Gains are in dB; negative values cut, positive values boost.
    # Typical CRT traits:
    #  - low_gain_db  slightly negative  (weak bass)
    #  - mid_gain_db  around 0 or positive (mid heavy voice)
    #  - high_gain_db negative (rolled off treble)
    "low_mid_split_hz": 300.0,
    "mid_high_split_hz": 3000.0,
 
    # Try:
    #   low_gain_db  = -6.0 to -12.0 for a tiny TV speaker
    #   mid_gain_db  = 0.0 to +4.0 for more "voice" presence
    #   high_gain_db = -6.0 to -18.0 for hiss free old TV top end
    "low_gain_db": -8.0,
    "mid_gain_db": +1.5,
    "high_gain_db": -12.0,
}
 
# ---------------------
# NOISE & HUM CONTROLS
# ---------------------
NOISE = {
    # HISS level in dB relative to the program signal RMS.
    #   -20 dB  : very audible hiss
    #   -30 dB  : moderate hiss
    #   -40 dB  : subtle hiss
    #   None    : disables hiss
    "hiss_level_db_rel": -32.0,
 
    # HUM level in dB relative to the program RMS.
    #   -25 dB  : strong 60 Hz hum
    #   -35 dB  : mild but noticeable hum
    #   -45 dB  : subtle hum
    #   None    : disables hum
    "hum_level_db_rel": -38.0,
 
    # Mains frequency for hum (60 in US, 50 in many other regions).
    "hum_frequency_hz": 60.0,
 
    # How many harmonics above the base hum to include.
    #  - More harmonics = buzzier/rougher hum.
    #  - 2–4 is usually enough for a CRT feel.
    "hum_harmonics": 3,
}
 
# ---------------------
# DISTORTION / GRIT
# ---------------------
DISTORTION = {
    # Drive in dB before a soft clipping stage (tanh distortion).
    #   0  dB : almost clean
    #   6  dB : mild TV speaker breakup
    #   12 dB : crunchy / clearly distorted
    "drive_db": 6.0,
 
    # Mix between clean and distorted signal (0.0–1.0).
    #   0.0 : no distortion
    #   0.5 : half clean, half dirty
    #   1.0 : fully distorted
    "mix": 0.7,
}
